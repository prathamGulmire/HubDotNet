CREATE TABLE Course
(
    CourseId INT IDENTITY(1,1) PRIMARY KEY,
    CourseCode VARCHAR(50) NOT NULL UNIQUE,
    CourseName VARCHAR(150) NOT NULL,
    Description VARCHAR(500) NULL,
    DurationMonths INT NOT NULL,
    IsActive BIT NOT NULL DEFAULT 1,
    CreatedAt DATETIME NOT NULL DEFAULT GETDATE(),
    UpdatedAt DATETIME NULL	
);

create table StudentCourse
(
	StudentCourseId int identity(1, 1) primary key,
	Sid int not null,
	Cid int not null,
	EnrolledOn datetime not null default getdate(),

	constraint FK_studentCourse_Student foreign key (Sid) references Mytable(id),
	constraint FK_studentCourse_Course foreign key (Cid) references Course(CourseId),

	constraint UQ_studentCourse unique (Sid, Cid)
);
